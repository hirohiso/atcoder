        Arrays.sort(p, (i, j) -> {
            final long jud1 = i.x * (j.x + j.y);
            final long jud2 = j.x * (i.x + i.y);
            if (jud1 == jud2)
                return Integer.compare(i.idx, j.idx);
            return Long.compare(jud2, jud1);
        });
